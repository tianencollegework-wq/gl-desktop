    /**
     * 这张表**真正**有哪些列；读不到时返回 null（不猜，也不瞎改）。
     */
    private function tableColumns($table)
    {
        try {
            $rows = Db::query('SHOW COLUMNS FROM `' . $table . '`');
        } catch (\Throwable $e) {
            trace('SFU：读注册表结构失败：' . $e->getMessage(), 'error');
            return null;
        }
        $names = [];
        foreach ((array)$rows as $row) {
            if (is_array($row) && isset($row['Field'])) {
                $names[] = (string)$row['Field'];
            }
        }
        return $names;
    }

    /**
     * 缺哪列补哪列（老结构的表就这么自愈）。返回"结构现在能不能用"。
     */
    private function ensureColumns($table)
    {
        $have = $this->tableColumns($table);
        if ($have === null) {
            // 结构都读不到（没权限？）：按可用处理，别把整条通话链路卡死
            return true;
        }
        if (!in_array('id', $have, true)) {
            // 主键列缺了没法自动补（AUTO_INCREMENT 得有键），只能报给运维
            trace('SFU：注册表缺主键列 id（这张表是老版本建的），请手工重建这张表', 'error');
            return false;
        }
        $missing = [];
        foreach ($this->requiredColumns() as $name => $ddl) {
            if (!in_array($name, $have, true)) {
                $missing[$name] = $ddl;
            }
        }
        foreach ($missing as $name => $ddl) {
            try {
                Db::execute('ALTER TABLE `' . $table . '` ADD COLUMN ' . $ddl);
                trace('SFU：注册表缺列 ' . $name . '（老结构），已自动补上', 'info');
            } catch (\Throwable $e) {
                trace(
                    'SFU：注册表缺列 ' . $name . ' 且自动补失败，请在数据库手工执行：'
                    . 'ALTER TABLE `' . $table . '` ADD COLUMN ' . $ddl
                    . ' —— ' . $e->getMessage(),
                    'error'
                );
            }
        }
        if (empty($missing)) {
            return true;
        }
        $now = $this->tableColumns($table);
        if (!is_array($now)) {
            return true;
        }
        return in_array('tracks', $now, true) && in_array('mids', $now, true);
    }

    private function ensureUniqueKey($table)
    {

            $rows = Db::query('SHOW INDEX FROM `' . $table . '`');

            return;

                return;
            }
        }
        try {
            Db::execute(
                'ALTER TABLE `' . $table . '` ADD UNIQUE KEY '
                . '`uk_room_user_purpose` (`room_id`,`user_id`,`purpose`)'
            );
            trace('SFU：注册表缺唯一键，已自动补上', 'info');
        } catch (\Throwable $e) {
            trace(
                'SFU：注册表缺唯一键且补失败（表里可能已有重复行，先删重复再执行）：'
                . 'ALTER TABLE `' . $table . '` ADD UNIQUE KEY '
                . '`uk_room_user_purpose` (`room_id`,`user_id`,`purpose`)'
                . ' —— ' . $e->getMessage(),
                'error'
            );
        }
    }
